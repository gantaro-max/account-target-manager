## 背景・目的

他担当者取引先参照機能（TASKS.md § 1.6）の実装後、担当者セレクターを切り替えるたびに
取引先1件ごとに `getAccountDetail` が呼ばれ、GASのオーバーレイが頻繁に表示される。

本改善では担当者選択時に全取引先詳細を **1回のGAS呼び出しで一括取得・キャッシュ** し、
以降の取引先切り替えをキャッシュヒット（即時表示）に変える。

## 1. GAS: `getAccountDetailBulk()` を追加

**場所**: `src/コード_v2_9.js` の `getAccountDetail` 関数の直前に挿入する。

```javascript
function getAccountDetailBulk(accountIds) {
  const email = Session.getActiveUser().getEmail();
  const tz = Session.getScriptTimeZone();

  const users = _getRows(SHEET.USERS);
  const me = users.find((u) => u.user_id === email) || {};
  const myRole = me.role || "member";
  if (myRole !== "leader" && myRole !== "member") {
    throw new Error("他担当者取引先の詳細参照権限がありません");
  }

  const leaderMemberEmails = new Set(
    users
      .filter((u) => u.role === "leader" || u.role === "member")
      .map((u) => u.user_id),
  );

  const allAccounts = _getRows(SHEET.ACCOUNTS).map(_mapAccount);

  // 権限のある ID だけに絞る（自分の取引先・他ロールの取引先は除外）
  const validIds = (accountIds || []).filter((id) => {
    const h = allAccounts.find((h) => h.id === id);
    return (
      h &&
      leaderMemberEmails.has(h.assignedUserId) &&
      h.assignedUserId !== email
    );
  });

  if (validIds.length === 0) return {};

  // 各シートを 1 回だけ読んで全取引先分まとめて返す
  const allDepts = _getRows(SHEET.DEPARTMENTS).map(_mapDepartment);
  const allProjects = _getRows(SHEET.PROJECTS).map((p) => _mapProject(p, tz));
  const allLogs = _getRows(SHEET.LOGS).map((l) => _mapLog(l, tz));

  const result = {};
  validIds.forEach((id) => {
    result[id] = {
      departments: allDepts.filter((d) => d.accountId === id),
      projects: allProjects.filter((p) => p.accountId === id),
      logs: allLogs.filter((l) => l.accountId === id),
    };
  });
  return result;
}
```

**ポイント**:
- `getAccountDetail`（単体）と同じ権限チェックロジックを踏襲する
- 各シートを 1 回だけ `getValues` することで呼び出しコストを抑える
- 権限外・自分の取引先は `validIds` から除外してサイレントにスキップ

## 3. `js.html`: フロントエンド側の改修

### 3-1. `prefetchOtherAccounts()` を新規追加

`ensureAccountDetail` 関数の直前に追加する。

```javascript
function prefetchOtherAccounts(userId) {
  if (!userId || userId === "self") return;

  const hospIds = (db.otherAccounts || [])
    .filter((h) => h.assignedUserId === userId)
    .map((h) => h.id);

  // 未キャッシュの取引先だけ対象にする
  const uncached = hospIds.filter((id) => !otherAccountCache[id]);
  if (uncached.length === 0) return;

  // Promise を先にセット（同期）→ changeAccount が ensureAccountDetail を呼ぶ前に参照可能にする
  _prefetchPromise[userId] = new Promise((resolve) => {
    google.script.run
      .withSuccessHandler((bulkDetail) => {
        Object.entries(bulkDetail || {}).forEach(([hospId, detail]) => {
          // すでに個別フェッチでキャッシュ済みの場合は上書きしない
          if (!otherAccountCache[hospId]) {
            otherAccountCache[hospId] = detail;
            mergeOtherAccountDetail(hospId, detail);
          }
        });
        delete _prefetchPromise[userId];
        resolve();
      })
      .withFailureHandler(() => {
        // バルク失敗はサイレントに無視し、個別フェッチのフォールバックへ委ねる
        delete _prefetchPromise[userId];
        resolve();
      })
      .getAccountDetailBulk(uncached);
  });
}
```

### 3-2. `ensureAccountDetail()` をリファクタ

既存の `ensureAccountDetail` を以下の実装に**丸ごと置き換える**。
バルクプリフェッチ中なら完了を待ち、個別GAS呼び出しを回避する。

```javascript
function ensureAccountDetail(accountId) {
  if (!accountId) return Promise.resolve();

  // キャッシュヒット: 即時
  if (otherAccountCache[accountId]) {
    mergeOtherAccountDetail(accountId, otherAccountCache[accountId]);
    return Promise.resolve();
  }

  // バルクプリフェッチ進行中なら完了を待つ（個別GAS呼び出しを回避）
  const userId = (db.otherAccounts || []).find((h) => h.id === accountId)
    ?.assignedUserId;
  const pending = userId ? _prefetchPromise[userId] : null;
  if (pending) {
    showOverlay("データを読み込んでいます...");
    return pending.then(() => {
      hideOverlay();
      if (otherAccountCache[accountId]) {
        mergeOtherAccountDetail(accountId, otherAccountCache[accountId]);
        return;
      }
      // バルクに含まれていなかった場合のフォールバック
      return _fetchAccountDetailSingle(accountId);
    });
  }

  // バルクなし・キャッシュなし → 個別フェッチ
  showOverlay("データを読み込んでいます...");
  return _fetchAccountDetailSingle(accountId);
}

function _fetchAccountDetailSingle(accountId) {
  return new Promise((resolve, reject) => {
    google.script.run
      .withSuccessHandler((detail) => {
        otherAccountCache[accountId] = detail || {};
        mergeOtherAccountDetail(accountId, otherAccountCache[accountId]);
        hideOverlay();
        resolve();
      })
      .withFailureHandler((e) => {
        hideOverlay();
        showError(e, "取引先詳細の取得に失敗しました");
        reject(e);
      })
      .getAccountDetail(accountId);
  });
}
```

### 3-3. `handleTargetUserChange()` を修正

`prefetchOtherAccounts(value)` の呼び出しを `changeAccount(1, t1HospId)` より**前**に置くことで、
T1 の先頭取引先自動選択がバルクの完了を待てるようにする。

```javascript
function handleTargetUserChange(tabIdx, value) {
  collectBasicInfo();
  if (tabIdx === 0) {
    currentTargetUserT0 = value || "self";
    t0HospId = "";
    if (typeof cancelEditLog === "function") cancelEditLog();
  }
  if (tabIdx === 1) {
    currentTargetUserT1 = value || "self";
    const accounts = getAccountsForTab(1);
    t1HospId = accounts.length > 0 ? accounts[0].id : "";
  }
  updateAccountSelectOptions();
  updateOtherUserReadonlyState();
  renderAll();
  // バルクプリフェッチを先に起動（_prefetchPromise をセット）してから changeAccount を呼ぶ
  if (value && value !== "self") prefetchOtherAccounts(value);
  if (tabIdx === 1 && t1HospId) changeAccount(1, t1HospId);
}
```

## 処理フロー図

```
担当者セレクター変更
  │
  ├─ prefetchOtherAccounts(userId) ← 同期的に _prefetchPromise[userId] をセット
  │    └─ GAS: getAccountDetailBulk([hospId1, hospId2, ...])  ← 非同期で開始
  │
  └─ changeAccount(1, t1HospId) ← T1 先頭取引先を選択
       └─ ensureAccountDetail(t1HospId)
            └─ _prefetchPromise[userId] が存在 → await して待機
                 ├─ バルク完了後 → キャッシュヒット → 即時描画
                 └─ バルク失敗 → _fetchAccountDetailSingle() フォールバック

以降の取引先切り替え（2回目〜）
  └─ ensureAccountDetail(hospId)
       └─ otherAccountCache ヒット → 即時描画（GAS呼び出しなし）
```
