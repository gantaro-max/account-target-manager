### 3.1 `getPlans()` — 全キャンペーン取得（管理用）

```js
function getPlans() {
  _requireManager();
  const sheet = _ss().getSheetByName(SHEET.CAMPAIGNS);
  if (!sheet) return [];
  const rows = sheet.getDataRange().getValues();
  if (rows.length <= 1) return [];
  const headers = rows[0];
  const idx = {
    id:          headers.indexOf("id"),
    name:        headers.indexOf("name"),
    description: headers.indexOf("description"),
    item:        headers.indexOf("item"),
    startDate:   headers.indexOf("start_date"),
    endDate:     headers.indexOf("end_date"),
  };
  return rows.slice(1)
    .filter(r => r[idx.id])
    .map(r => ({
      id:          String(r[idx.id] || ""),
      name:        String(r[idx.name] || ""),
      description: String(r[idx.description] || ""),
      item:        String(r[idx.item] || ""),
      startDate:   r[idx.startDate] ? Utilities.formatDate(new Date(r[idx.startDate]), Session.getScriptTimeZone(), "yyyy-MM-dd") : "",
      endDate:     r[idx.endDate]   ? Utilities.formatDate(new Date(r[idx.endDate]),   Session.getScriptTimeZone(), "yyyy-MM-dd") : "",
    }));
}
```

### 3.2 `saveCampaign(campaign)` — 新規追加 or 更新

```js
function saveCampaign(campaign) {
  _requireManager();
  if (campaign.id) {
    _updateById(SHEET.CAMPAIGNS, "id", campaign.id, {
      name:        campaign.name || "",
      description: campaign.description || "",
      item:        campaign.item || "",
      start_date:  campaign.startDate || "",
      end_date:    campaign.endDate || "",
    });
    return { ok: true, id: campaign.id };
  } else {
    const newId = "pl" + Date.now();
    _appendRow(SHEET.CAMPAIGNS, {
      id:          newId,
      name:        campaign.name || "",
      description: campaign.description || "",
      item:        campaign.item || "",
      start_date:  campaign.startDate || "",
      end_date:    campaign.endDate || "",
    });
    return { ok: true, id: newId };
  }
}
```

### 3.3 `deleteCampaign(campaignId)` — 削除

```js
function deleteCampaign(campaignId) {
  _requireManager();
  return { ok: _deleteById(SHEET.CAMPAIGNS, "id", campaignId) };
}
```

**注意**: 削除しても既存の `projects.name` フィールドには自由テキストとして値が残る。参照整合性の心配は不要。

## 4. `js.html` の修正（`init()` 関数内）

`getActivePlans()` の呼び出しブロック（186行付近）の**直後**に以下を追加する。  
`db.campaigns` は管理者・責任者のみ使用するため、ロールチェック後に呼び出す。

```js
// キャンペーンマスター管理用（admin/manager のみ）
if (currentUserRole === "admin" || currentUserRole === "manager") {
  google.script.run
    .withSuccessHandler(function (campaigns) {
      db.campaigns = campaigns || [];
      if (typeof renderCampaignList === "function") renderCampaignList();
    })
    .withFailureHandler(function (e) {
      console.warn("キャンペーンリスト(管理用)取得失敗:", e);
      db.campaigns = [];
    })
    .getPlans();
}
```

ただし `currentUserRole` は `withSuccessHandler` コールバック内でのみ確定する。  
上記のブロックは `serverData` 取得成功後のコールバック内（`currentUserRole` 確定後）に記述すること。  
→ 具体的には、`switchTab(0)` を呼ぶ直前あたりに配置する。

## 6. `js_review.html` の修正

ファイル末尾の `</script>` の直前に以下を追加する。

```js
// ============================================================
// キャンペーンマスター管理
// ============================================================

function _getCampaignStatus(campaign) {
  const today = new Date(); today.setHours(0, 0, 0, 0);
  const start = campaign.startDate ? new Date(campaign.startDate) : null;
  const end   = campaign.endDate   ? new Date(campaign.endDate)   : null;
  if (start && start > today) return { label: "開始前", color: "#64748b" };
  if (end   && end   < today) return { label: "終了済み", color: "#94a3b8" };
  return { label: "期間中", color: "var(--teal)" };
}

function renderCampaignList() {
  const area = $("campaign-list-area");
  if (!area) return;
  const campaigns = db.campaigns || [];
  if (campaigns.length === 0) {
    area.innerHTML = '<div class="section-note">キャンペーンが登録されていません</div>';
    return;
  }
  area.innerHTML = `
    <table style="width:100%; border-collapse:collapse; font-size:13px;">
      <thead>
        <tr style="background:var(--light-bg,#f8fafc); border-bottom:2px solid var(--border);">
          <th style="text-align:left; padding:8px 6px; font-weight:600;">キャンペーン名</th>
          <th style="text-align:left; padding:8px 6px; font-weight:600;">対象製品</th>
          <th style="text-align:left; padding:8px 6px; font-weight:600;">期間</th>
          <th style="text-align:center; padding:8px 6px; font-weight:600;">状態</th>
          <th style="padding:8px 6px;"></th>
        </tr>
      </thead>
      <tbody>
        ${campaigns.map(p => {
          const st = _getCampaignStatus(p);
          const period = [p.startDate || "未設定", p.endDate || "未設定"].join(" 〜 ");
          return `
            <tr style="border-bottom:1px solid var(--border);">
              <td style="padding:8px 6px; font-weight:500; color:var(--navy);">${p.name}</td>
              <td style="padding:8px 6px; color:var(--gray);">${p.item || "—"}</td>
              <td style="padding:8px 6px; color:var(--gray); font-size:12px;">${period}</td>
              <td style="padding:8px 6px; text-align:center;">
                <span style="
                  display:inline-block;
                  padding:2px 8px;
                  border-radius:12px;
                  font-size:11px;
                  font-weight:600;
                  color:white;
                  background:${st.color};
                ">${st.label}</span>
              </td>
              <td style="padding:8px 6px; text-align:right; white-space:nowrap;">
                <button class="control-btn" style="font-size:11px; padding:2px 8px; margin-right:4px;" onclick="openCampaignEdit('${p.id}')">編集</button>
                <button class="del-btn" style="font-size:11px; padding:2px 8px; color:var(--red); border-color:var(--red);" onclick="deleteCampaignItem('${p.id}')">削除</button>
              </td>
            </tr>
          `;
        }).join("")}
      </tbody>
    </table>
  `;
}

function openCampaignEdit(campaignId) {
  const campaign = (db.campaigns || []).find(p => p.id === campaignId);
  if (!campaign) return;
  $("campaign-edit-id").value = campaign.id;
  $("campaign-name").value = campaign.name;
  $("campaign-item").value = campaign.item || "";
  $("campaign-description").value = campaign.description || "";
  $("campaign-start-date").value = campaign.startDate || "";
  $("campaign-end-date").value = campaign.endDate || "";
  $("campaign-form-title").textContent = "✏️ キャンペーンを編集中";
  $("campaign-cancel-btn").style.display = "";
  $("campaign-form-area").scrollIntoView({ behavior: "smooth", block: "nearest" });
}

function cancelCampaignEdit() {
  $("campaign-edit-id").value = "";
  $("campaign-name").value = "";
  $("campaign-item").value = "";
  $("campaign-description").value = "";
  $("campaign-start-date").value = "";
  $("campaign-end-date").value = "";
  $("campaign-form-title").textContent = "➕ 新規キャンペーンを追加";
  $("campaign-cancel-btn").style.display = "none";
}

function saveCampaignItem() {
  const name = ($("campaign-name").value || "").trim();
  if (!name) {
    Swal.fire({ icon: "warning", title: "キャンペーン名は必須です", timer: 2000, showConfirmButton: false });
    return;
  }
  const campaign = {
    id:          $("campaign-edit-id").value || "",
    name:        name,
    item:        ($("campaign-item").value || "").trim(),
    description: ($("campaign-description").value || "").trim(),
    startDate:   $("campaign-start-date").value || "",
    endDate:     $("campaign-end-date").value || "",
  };
  showSaving();
  google.script.run
    .withSuccessHandler(function (res) {
      if (!campaign.id) {
        campaign.id = res.id;
        (db.campaigns || (db.campaigns = [])).push(campaign);
      } else {
        const idx = (db.campaigns || []).findIndex(p => p.id === campaign.id);
        if (idx >= 0) db.campaigns[idx] = campaign;
      }
      // datalist も更新（期間中ならactivePlansに追加）
      if (typeof populateCampaignDatalist === "function") {
        const today = new Date(); today.setHours(0,0,0,0);
        const start = campaign.startDate ? new Date(campaign.startDate) : null;
        const end   = campaign.endDate   ? new Date(campaign.endDate)   : null;
        const isActive = (!start || start <= today) && (!end || end >= today);
        db.activePlans = db.activePlans || [];
        const aidx = db.activePlans.findIndex(p => p.name === campaign.name);
        if (isActive && aidx < 0) {
          db.activePlans.push({ name: campaign.name, description: campaign.description, item: campaign.item });
        } else if (!isActive && aidx >= 0) {
          db.activePlans.splice(aidx, 1);
        } else if (isActive && aidx >= 0) {
          db.activePlans[aidx] = { name: campaign.name, description: campaign.description, item: campaign.item };
        }
        populateCampaignDatalist();
      }
      cancelCampaignEdit();
      renderCampaignList();
      showSuccess();
    })
    .withFailureHandler(function (e) {
      showError(e, "キャンペーンの保存に失敗しました");
    })
    .saveCampaign(campaign);
}

function deleteCampaignItem(campaignId) {
  const campaign = (db.campaigns || []).find(p => p.id === campaignId);
  if (!campaign) return;
  showConfirm(
    `「${campaign.name}」を削除しますか？\n※過去の案件に登録済みのキャンペーン名はそのまま残ります。`,
    "キャンペーンの削除確認",
    function () {
      showSaving();
      google.script.run
        .withSuccessHandler(function () {
          db.campaigns = (db.campaigns || []).filter(p => p.id !== campaignId);
          db.activePlans = (db.activePlans || []).filter(p => p.name !== campaign.name);
          if (typeof populateCampaignDatalist === "function") populateCampaignDatalist();
          renderCampaignList();
          showSuccess();
        })
        .withFailureHandler(function (e) {
          showError(e, "キャンペーンの削除に失敗しました");
        })
        .deleteCampaign(campaignId);
    },
    "削除する"
  );
}
```

## 7. データフロー

```
[起動時]
  getActivePlans() → db.activePlans  → 案件フォームの <datalist> に表示（期間中のみ）
  getPlans()       → db.campaigns        → キャンペーン管理画面の一覧テーブルに表示（全件）

[保存後]
  db.campaigns を更新 → renderCampaignList() で一覧再描画
  db.activePlans も同期更新 → populateCampaignDatalist() でdatalist再生成
```

## 8. UI仕様まとめ

- **場所**: ⚙️ 設問管理タブ（`id="tab4"`）の設問管理セクションの**下**に区切り線を挟んで追加
- **表示権限**: `admin` / `manager`（`.admin-only` クラス制御により既に両ロールへ表示済み）
- **状態バッジ**: 開始前=グレー / 期間中=ティール / 終了済み=薄グレー
- **編集フォーム**: 上部の追加フォームを兼用（編集時はタイトルが「✏️ 編集中」に変化し、キャンセルボタンが現れる）
- **削除**: `showConfirm()` で確認後に物理削除。案件への参照影響なし（案件の `name` は自由テキスト）
