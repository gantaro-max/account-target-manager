# 実装指示書: 案件進捗レポート CSV出力機能

**作成日**: 2026-06-09
**対象ブランチ**: feature/advance

---

## 背景・目的

案件名（キャンペーン名）をキーワードに絞り込んで、担当者・取引先情報・進捗ステージ・備考・活動履歴を
1件の CSV としてダウンロードする機能を追加する。
主にマネージャーがキャンペーン単位の進捗を横断的に把握するために使用する。

## 1. `src/コード.js` の変更

### 追加場所

`getCsvData()` 関数の直後（`getStatistics()` の前）に挿入する。

### 追加する関数

```javascript
function getProjectReport(nameQuery) {
  _requireManager();
  const tz = Session.getScriptTimeZone();

  const accounts  = _getRows(SHEET.ACCOUNTS);
  const projects   = _getRows(SHEET.PROJECTS);
  const logs       = _getRows(SHEET.LOGS);
  const users      = _getRows(SHEET.USERS);
  const masterList = _getRows(SHEET.MASTER);

  const accountMap = {};
  accounts.forEach(function(h) { accountMap[h.id] = h; });

  const userMap = {};
  users.forEach(function(u) { userMap[u.user_id] = u; });

  const masterMap = {};
  masterList.forEach(function(m) {
    const code = String(m.code || "").trim();
    if (code) masterMap[code] = m;
  });

  const STAGE_LABELS = [
    "0：未接触",
    "1：初回訪問済",
    "2：ニーズ確認中",
    "3：提案実施中",
    "4：クロージング",
  ];

  function normalizeStr(str) {
    return String(str || "").normalize("NFKC").toLowerCase();
  }

  function formatDate(value) {
    if (!value) return "";
    if (Object.prototype.toString.call(value) === "[object Date]" && !isNaN(value.getTime())) {
      return Utilities.formatDate(value, tz, "yyyy-MM-dd");
    }
    const s = String(value).trim();
    if (/^\d{4}-\d{2}-\d{2}/.test(s)) return s.slice(0, 10);
    const d = new Date(value);
    return isNaN(d.getTime()) ? s : Utilities.formatDate(d, tz, "yyyy-MM-dd");
  }

  const normalizedQuery = normalizeStr(nameQuery || "");
  if (!normalizedQuery) return null;

  const matchedProjects = projects.filter(function(p) {
    return normalizeStr(p.name).indexOf(normalizedQuery) !== -1;
  });

  if (matchedProjects.length === 0) return null;

  // project_id → logs[] を visit_date 降順で保持
  const logsByProject = {};
  logs.forEach(function(l) {
    if (!logsByProject[l.project_id]) logsByProject[l.project_id] = [];
    logsByProject[l.project_id].push(l);
  });
  Object.keys(logsByProject).forEach(function(pid) {
    logsByProject[pid].sort(function(a, b) {
      return String(b.visit_date || "") < String(a.visit_date || "") ? -1 : 1;
    });
  });

  const headers = [
    "担当者名",
    "code",
    "拠点名",
    "パートナー名",
    "取引先名",
    "住所",
    "案件名",
    "進捗ステージ",
    "備考",
    "訪問日",
    "活動内容",
  ];

  const rows = [];
  matchedProjects.forEach(function(project) {
    const account     = accountMap[project.account_id] || {};
    const user         = userMap[account.assigned_user_id] || {};
    const accountCode = String(account.code || "").trim();
    const master       = (accountCode ? masterMap[accountCode] : null) || {};
    const stageNum     = Number(project.current_stage);
    const stageLabel   = STAGE_LABELS[stageNum] !== undefined
      ? STAGE_LABELS[stageNum]
      : String(project.current_stage || "");

    const base = [
      user.name         || "",
      account.code     || "",
      master.office     || "",
      account.partner_name  || "",
      account.name     || "",
      String(account.addr || "").trim() || master.address || "",
      project.name      || "",
      stageLabel,
      project.latest_note || "",
    ];

    const projectLogs = logsByProject[project.id] || [];
    if (projectLogs.length === 0) {
      rows.push(base.concat(["", ""]));
    } else {
      projectLogs.forEach(function(log) {
        rows.push(base.concat([
          formatDate(log.visit_date),
          log.content || "",
        ]));
      });
    }
  });

  return _buildCsv(headers, rows);
}
```

### ポイント

- `_requireManager()` でマネージャー以上のみ実行可能とする。
- `masterMap` は `String(code).trim()` をキーにする。
- 住所は `account.addr` を優先し、空の場合に `master.address` にフォールバックする。
- 案件のログが 0 件でも案件行は出力する（訪問日・活動内容は空白）。
- 結果が 0 件の場合は `null` を返す（クライアント側で「該当なし」ダイアログを表示）。
- 戻り値は `_buildCsv()` を使った BOM 付き UTF-8 CSV 文字列（既存の `getCsvData` と同形式）。

```javascript
// 変更前
function populateCampaignDatalist() {
  const datalist = $("campaign-datalist");
  if (!datalist) return;
  datalist.innerHTML = "";
  (db.activePlans || []).forEach(function (campaign) {
    const option = document.createElement("option");
    option.value = campaign.name;
    if (campaign.description) option.label = campaign.description;
    datalist.appendChild(option);
  });
}

// 変更後
function populateCampaignDatalist() {
  ["campaign-datalist", "report-campaign-datalist"].forEach(function(id) {
    const datalist = $(id);
    if (!datalist) return;
    datalist.innerHTML = "";
    (db.activePlans || []).forEach(function(campaign) {
      const option = document.createElement("option");
      option.value = campaign.name;
      if (campaign.description) option.label = campaign.description;
      datalist.appendChild(option);
    });
  });
}
```

> **備考**: datalist に表示されるのは `db.activePlans`（期間中のキャンペーン）のみ。

## 4. `src/js_log.html` の変更

### 追加箇所

既存の `downloadCsv()` 関数の **直後**（`</script>` タグの前）に以下を追加する。

```javascript
  function downloadProjectReportCsv() {
    const query = ($("report-query").value || "").trim();
    if (!query) {
      Swal.fire("入力エラー", "案件名・キャンペーン名を入力してください。", "warning");
      return;
    }
    Swal.fire({
      title: "CSV出力中...",
      allowOutsideClick: false,
      didOpen: () => Swal.showLoading(),
    });
    google.script.run
      .withSuccessHandler(function(csv) {
        Swal.close();
        if (!csv) {
          Swal.fire("該当なし", "条件に一致する案件が見つかりませんでした。", "info");
          return;
        }
        const blob = new Blob([csv], { type: "text/csv;charset=utf-8;" });
        const url  = URL.createObjectURL(blob);
        const a    = document.createElement("a");
        a.href     = url;
        a.download = "案件進捗レポート_" + query + "_" + new Date().toISOString().slice(0, 10) + ".csv";
        document.body.appendChild(a);
        a.click();
        setTimeout(function() {
          document.body.removeChild(a);
          URL.revokeObjectURL(url);
        }, 100);
      })
      .withFailureHandler(function(e) {
        Swal.fire("エラー", e.message, "error");
      })
      .getProjectReport(query);
  }
```

## CSV 出力仕様

| # | ヘッダー | 取得元 | 備考 |
| :---: | :--- | :--- | :--- |
| 1 | 担当者名 | `users.name` | `account.assigned_user_id` から引く |
| 2 | code | `accounts.code` | 空欄の場合は空白 |
| 3 | 拠点名 | `master_list.office` | `accounts.code` で JOIN、未一致は空白 |
| 4 | パートナー名 | `accounts.partner_name` | 空欄の場合は空白 |
| 5 | 取引先名 | `accounts.name` | |
| 6 | 住所 | `accounts.addr` | 空の場合は `master_list.address` にフォールバック |
| 7 | 案件名 | `projects.name` | |
| 8 | 進捗ステージ | `projects.current_stage` | `STAGE_LABELS` のラベル文字列で出力 |
| 9 | 備考 | `projects.latest_note` | 空欄の場合は空白 |
| 10 | 訪問日 | `activity_logs.visit_date` | `yyyy-MM-dd` 形式、降順 |
| 11 | 活動内容 | `activity_logs.content` | |

- **フォーマット**: BOM 付き UTF-8 CSV（Excel で文字化けなし）
- **行構造**: 1行 = 1活動ログ。案件情報は各行に繰り返す。活動ログ 0 件の案件も 1 行出力。
- **ファイル名**: `案件進捗レポート_{入力クエリ}_{YYYY-MM-DD}.csv`
